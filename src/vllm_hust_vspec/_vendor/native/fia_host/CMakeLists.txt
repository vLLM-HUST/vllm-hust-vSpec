cmake_minimum_required(VERSION 3.18)
project(vspec_fia_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Torch REQUIRED)
find_package(Python3 3.11 EXACT COMPONENTS Interpreter Development REQUIRED)
set(CANN "$ENV{ASCEND_HOME_PATH}" CACHE PATH "Installed CANN SDK")
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import os, torch, torch_npu; print(os.path.dirname(torch.__file__)); print(os.path.dirname(torch_npu.__file__))"
    OUTPUT_VARIABLE PYTHON_PATHS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
string(REPLACE "\n" ";" PYTHON_PATHS "${PYTHON_PATHS}")
list(GET PYTHON_PATHS 0 TORCH_PATH)
list(GET PYTHON_PATHS 1 TORCH_NPU_PATH)
add_library(vspec_fia_host MODULE adapter.cpp)
set_target_properties(vspec_fia_host PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
target_include_directories(vspec_fia_host PRIVATE ${TORCH_INCLUDE_DIRS} ${Python3_INCLUDE_DIRS} "${TORCH_NPU_PATH}/include" "${CANN}/include")
target_link_directories(vspec_fia_host PRIVATE "${TORCH_PATH}/lib" "${TORCH_NPU_PATH}/lib" "${CANN}/lib64")
target_link_libraries(vspec_fia_host PRIVATE ${TORCH_LIBRARIES} torch_python torch_npu ascendcl Python3::Python)
set_target_properties(vspec_fia_host PROPERTIES BUILD_RPATH "${TORCH_PATH}/lib;${TORCH_NPU_PATH}/lib;${CANN}/lib64")
